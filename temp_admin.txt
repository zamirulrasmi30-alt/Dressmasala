// ---------- Date filtering ----------
// Sheets locale is DD/MM/YYYY, but GAS may also return ISO strings â€”
// orderDayKey() normalises every supported shape to a local YYYY-MM-DD
// key so exact-date and range comparisons are simple string compares.
const toDayKey = (d) =>
  `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(
    d.getDate()
  ).padStart(2, '0')}`;

const orderDayKey = (o) => {
  const v = o?.date;
  if (!v) return '';
  if (typeof v === 'string') {
    const iso = v.match(/^(\d{4})-(\d{2})-(\d{2})/);
    if (iso) return `${iso[1]}-${iso[2]}-${iso[3]}`;
    // DD/MM/YYYY or DD-MM-YYYY (also tolerates 2-digit year).
    const dmy = v.match(/^(\d{1,2})[\/\-.](\d{1,2})[\/\-.](\d{2,4})/);
    if (dmy) {
      let [, d, mo, y] = dmy;
      if (y.length === 2) y = '20' + y;
      return `${y}-${mo.padStart(2, '0')}-${d.padStart(2, '0')}`;
    }
  }
  const t = new Date(v).getTime();
  if (Number.isNaN(t)) return '';
  return toDayKey(new Date(t));
};

const fmtDay = (key) => {
  const m = String(key).match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (!m) return key;
  return new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3])).toLocaleDateString(
    'en-IN',
    { day: 'numeric', month: 'short' }
  );
};

  const [preset, setPreset] = useState('all');
  const [from, setFrom] = useState('');
  const [to, setTo] = useState('');

  const applyPreset = (p) => {
    const now = new Date();
    setPreset(p);
    if (p === 'today') {
      const k = toDayKey(now);
      setFrom(k);
      setTo(k);
    } else if (p === 'yesterday') {
      const k = toDayKey(new Date(now.getTime() - 86400000));
      setFrom(k);
      setTo(k);
    } else if (p === 'week') {
      setFrom(toDayKey(new Date(now.getTime() - 6 * 86400000)));
      setTo(toDayKey(now));
    } else {
      setFrom('');
      setTo('');
    }
  };

  // Orders visible under the active date filter (auto-swaps flipped bounds).
  const visibleOrders = useMemo(() => {
    if (preset === 'all' || !from || !to) return orders;
    const [a, b] = from <= to ? [from, to] : [to, from];
    return orders.filter((o) => {
      const k = orderDayKey(o);
      return k && k >= a && k <= b;
    });
  }, [orders, preset, from, to]);

  const rangeLabel =
    preset === 'all' ? '' : from === to ? fmtDay(from) : `${fmtDay(from)} â€“ ${fmtDay(to)}`;

  const stats = useMemo(
    () => computeOverview(products, orders, subscribers),
    [products, orders, subscribers]
  );

  // Week-over-week trends (last 7 days vs the 7 before) for the

  const trends = useMemo(() => {
    const now = Date.now();
    const W = 7 * 86400000;
    const inRange = (o, from, to) => {
      const t = new Date(o.date).getTime();
      return !Number.isNaN(t) && t >= from && t < to;
    };
    const cur = orders.filter((o) => inRange(o, now - W, now));
    const prev = orders.filter((o) => inRange(o, now - 2 * W, now - W));
    const revenue = (arr) => arr.reduce((s, o) => s + (Number(o.total) || 0), 0);
    const pct = (c, p) => {
      if (p > 0) return ((c - p) / p) * 100;
      return c > 0 ? 100 : NaN;
    };
    return { revPct: pct(revenue(cur), revenue(prev)), ordPct: pct(cur.length, prev.length) };
  }, [orders]);

  const trendPill = (pctValue) => {
    if (!Number.isFinite(pctValue)) {
      return <span className="trend flat">Â· steady</span>;
    }
    const up = pctValue >= 0;
    return (
      <span className={`trend ${up ? 'up' : 'down'}`}>
        {up ? 'â–²' : 'â–¼'} {Math.abs(pctValue).toFixed(1)}% this week
      </span>
    );
  };

  // Metrics follow the date filter: revenue / active / totals recompute
  // from visibleOrders while inventory + subscribers stay global.
  const viewStats = useMemo(
    () =>
      preset === 'all' ? stats : computeOverview(products, visibleOrders, subscribers),
    [preset, stats, products, visibleOrders, subscribers]
  );

  const flash = (msg) => {
    setNotice(msg);
    window.clearTimeout(flash._t);
    flash._t = window.setTimeout(() => setNotice(''), 2600);
  };

  // Last-7-days revenue bars for the mini chart.
  const bars = useMemo(() => {

      const d = new Date();
      d.setDate(d.getDate() - (6 - i));
      return { key: d.toISOString().slice(0, 10), label: d.toLocaleDateString('en-IN', { weekday: 'short' }), total: 0 };
    });

      const k = String(o.date || '').slice(0, 10);
      const b = days.find((d) => d.key === k);
      if (b) b.total += Number(o.total) || 0;
    });

    });

  // Multi-file upload: fills the first empty views in order (max 3 views).
  const onFiles = async (e) => {

    if (!files.length) return;
    try {
      const urls = [];
      for (const file of files) urls.push(await fileToDataUrl(file));
      let placed = 0;
      setForm((f) => {

          String((Array.isArray(f.images) ? f.images[j] : '') || '')

        urls.forEach((url) => {

        return { ...f, images: next.slice(0, 3) };
      });

        placed

    } catch (err) {
      setFormErr(err.message);
    }
    e.target.value = '';
  };

  const submitForm = async (e) => {
    e.preventDefault();
    const err = validateProduct(form);
    if (err) {
      setFormErr(err);
      return;
    }
    setBusy(true);
    try {
      // Explicit 3-view payload: trimmed per-slot array PLUS the

      // backend stores, all 3 views survive the round trip.
      const views = [0, 1, 2].map((i) => String(form.images?.[i] || '').trim());

    } finally {

    flash(`â€œ${p.name}â€ deleted.`);

  const goTab = (id) => {

    setSideOpen(false);
  };

  /* ---------- render ---------- */

  return (
    <div className="admin">
      <div className="ad-ambient" aria-hidden="true" />
      {sideOpen && <div className="ad-scrim" onClick={() => setSideOpen(false)} />}

      {/* Sidebar */}
      <aside className={`ad-side ${sideOpen ? 'open' : ''}`} aria-label="Admin navigation">
        <div className="ad-brand">
          â™› ZAREEF<small>ADMIN</small>
        </div>
        <p className="ad-nav-label">Manage</p>
        {TABS.map((t) => (
          <button
            key={t.id}
            className={`ad-link ${tab === t.id ? 'active' : ''}`}

          â† Back to Website

          <span className={`ad-dot ${source === 'gas' ? 'live' : ''}`} />
          {source === 'gas' ? 'Live Â· Google Sheets' : 'Local mode Â· connect GAS'}
        </p>
      </aside>

      {/* Main */}
      <div className="ad-main">
        <div className="ad-top">
          <button className="ad-hamburger" aria-label="Open menu" onClick={() => setSideOpen(true)}>
            â˜°
          </button>
          <div>
            <h1>
              {tab === 'overview' && 'Dashboard Overview'}
              {tab === 'products' && 'Manage Dresses'}
              {tab === 'orders' && 'Orders'}
              {tab === 'subscribers' && 'Subscribers'}
            </h1>
            <p>
              {loading
                ? 'Syncing with Google Sheetsâ€¦'
                : `${products.length} products Â· ${orders.length} orders Â· ${subscribers.length} subscribers`}

          </div>
          <div className="ad-search">
            {(tab === 'products' || tab === 'orders') && (
              <input
                placeholder={tab === 'products' ? 'Search products or SKUâ€¦' : 'Search ordersâ€¦'}
                value={query}
                onChange={(e) => setQuery(e.target.value)}
                aria-label="Search"
              />
            )}
            <button className="ad-btn ghost" onClick={onRefresh}>
              âŸ³ Refresh
            </button>
            <button className="ad-btn danger" onClick={onLogout} title="Log out and return home">
              â» Logout
            </button>
            {tab === 'products' && (
              <button className="ad-btn" onClick={openAdd}>
                + Add Product

        </div>

        {/* Date filter â€” drives Overview metrics + Orders list */}

                <div className="trend-row">
                  {preset === 'all' ? (

                      <td>

                          <s>{inr(p.mrp)}</s>

                      </td>
                      <td>{p.stock}</td>
                      <td>
                        <div className="row-actions">
                          <button onClick={() => openEdit(p)}>Edit</button>

                          </button>
                        </div>
                      </td>
                    </tr>
                  ))}
                  {filteredProducts.length === 0 && (
                    <tr>
                      <td colSpan={6} className="ad-empty">
                        No products match â€œ{query}â€.
                      </td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          </div>
          </>
        )}

        {/* ===== ORDERS ===== */}
        {tab === 'orders' && (
          <div className="ad-table-wrap">
            <div className="ad-table-scroll">
              <table className="ad-table">

                  <tr>

                  </tr>
                </thead>
                <tbody>
                  {visibleOrders
                    .filter((o) =>

                    .map((o) => (
                      <tr key={o.orderId}>
                        <td>
                          <strong>{o.orderId}</strong>
                          <br />

                      </tr>
                    ))}
                  {visibleOrders.length === 0 && (
                    <tr>
                      <td colSpan={5} className="ad-empty">
                        No orders received on this date.
                      </td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* ===== SUBSCRIBERS ===== */}
        {tab === 'subscribers' && (
          <div className="ad-table-wrap">
            <div className="ad-table-scroll">
              <table className="ad-table">

                    <tr>
                      <td colSpan={2} className="ad-empty">
                        No subscribers yet â€” newsletter signups appear here live.
                      </td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          </div>
        )}
      </div>

      {/* Product add/edit modal */}
      {showForm && (
        <div className="ad-modal-back" onClick={() => !busy && setShowForm(false)}>

                    required

                    <option value="">â€” Select color â€”</option>
                    {COLOR_OPTIONS.map((c) => (
                      <option key={c} value={c}>
                        {c}
                      </option>
                    ))}
                  </select>
                </div>
                <div className="ad-field full">
                  <label>Images * â€” Main, Design Detail, Logo / Tag</label>
                  {['Main Image URL (1st View) *', 'Design Detail Image URL (2nd View)', 'Logo / Tag Image URL (3rd View)'].map(
                    (label, i) => (
                      <span key={label} className="ad-img-row" style={{ marginBottom: 10 }}>
                        {(form.images[i] || '').trim() ? (
                          <img

                            src={form.images[i]}

                      </span>

                  <input

                <button
                  type="button"

                <button type="submit" className="ad-btn" disabled={busy}>

    </div>
  );
